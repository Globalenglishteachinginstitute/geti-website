```javascript function searchCertificates(type, value) { var sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(CERTIFICATE_SHEET); if (!sheet) { return { success: false, message: "Certificate database was not found." }; } var data = sheet .getDataRange() .getDisplayValues(); if (data.length <= 1) { return { success: false, message: "No certificate records found." }; } var search = String(value || "") .trim() .toLowerCase(); if (!search) { return { success: false, message: "Please enter a name, certificate number, or search letter." }; } var results = []; for (var i = 1; i < data.length; i++) { var name = String(data[i][0] || "") .trim() .toLowerCase(); var nationality = String(data[i][1] || "") .trim() .toLowerCase(); var certificate = String(data[i][2] || "") .trim() .toLowerCase(); var certificateId = String(data[i][3] || "") .trim() .toLowerCase(); var certificateType = String(data[i][4] || "") .trim() .toLowerCase(); var found = false; /* * NAME SEARCH * * Example: * "r" -> finds Rakoto, Berta, Aldo... * "ra" -> finds names containing "ra" * "rakoto" -> finds Rakoto... */ if (type === "name") { found = name.indexOf(search) !== -1; } /* * CERTIFICATE ID SEARCH */ else if (type === "id") { found = certificateId.indexOf(search) !== -1; } /* * NATIONALITY SEARCH */ else if (type === "nationality") { found = nationality.indexOf(search) !== -1; } /* * GENERAL SEARCH * * Searches all important fields. */ else if (type === "all") { found = name.indexOf(search) !== -1 || nationality.indexOf(search) !== -1 || certificate.indexOf(search) !== -1 || certificateId.indexOf(search) !== -1 || certificateType.indexOf(search) !== -1; } if (found) { results.push({ name: data[i][0] || "", nationality: data[i][1] || "", certificate: data[i][2] || "", id: data[i][3] || "", type: data[i][4] || "" }); } } return { success: true, count: results.length, results: results }; } /* * VERIFY ONE CERTIFICATE */ function verifyCertificate(value) { return searchCertificates( "id", value ); } ```
